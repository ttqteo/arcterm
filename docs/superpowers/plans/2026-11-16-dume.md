> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** When the 5-hour Claude window is about to reset with quota left over, wavesrv spends it on initiative chunks
the person handed to DUM-E (owner `dum-e`), one sonnet quick run per chunk, each committing on its own branch and never
landing.

**Architecture:** A pure package `pkg/dume` decides (gate: start or not, and why) and reads the queue off the
initiatives (chunk state is kept in the chunk's status and `DUM-E: …` notes, so nothing else is stored). A wavesrv loop
(`wshserver_dume.go`) ticks every 5 minutes: it finishes chunks whose run ended (branch ahead of HEAD → "ready to land",
else blocked), asks the gate, and starts the next chunk in a fresh `wave/dume-…` worktree. `dume:enabled` (off by
default) turns it on; `wsh dume status` and a Brief header line say what it did and why.

## Decided (do not re-open)

- Queue: chunks with `owner == "dum-e"` and status `pending`, in active initiatives that have a project with a path
  (`wconfig` `Projects[name].Path`), initiative order then chunk order. One DUM-E run at a time.
- Each chunk: quick run, runtime `claude`, model `sonnet`, in a worktree on branch `wave/dume-<effort8>-<base36 unix>`
  made from the project's HEAD. The run commits there; nothing merges, pushes or lands. The chunk is marked done only
  by the person, after they land the branch.
- Start a window only when **all** hold: ≤ 60 min to the 5h reset; ≥ 30 % of the 5h window left (used ≤ 70 %); week
  used ≤ even pace − 10 points; none of the person's own runs live; memory pressure not warn/critical.
- Once a window is started ("armed" on that window's reset time), keep taking chunks until 90 % used or ≤ 10 min to
  the reset; the week, own-runs and pressure checks still apply to each new chunk.
- Week reading: `WeekPct`/`WeekReset`; when the account has none (a Team seat), the highest `ScopedWeeks` entry;
  with neither, DUM-E does not start ("no weekly reading").
- Even pace: $\text{pace} = 100 \cdot \dfrac{now - (weekReset - 7d)}{7d}$, clamped to $[0, 100]$.

```go
// Copyright 2026, Command Line Inc.
// SPDX-License-Identifier: Apache-2.0

// Package dume is DUM-E: it spends the 5-hour Claude window's leftover quota, just before the window resets, on the
// initiative chunks the person handed it (owner "dum-e"). This file is the gate: given the quota, the clock and what
// else is running, start the next chunk or not, and say why in words the cockpit shows. Pure.
package dume

import (
	"fmt"
	"time"

	"github.com/wavetermdev/waveterm/pkg/claudequota"
	"github.com/wavetermdev/waveterm/pkg/jobqueue"
)

const (
	startWithin     = 60 * time.Minute // a window starts only this close to its reset
	startMaxUsedPct = 70.0             // ... with at least 30 % of it left
	stopAtUsedPct   = 90.0             // a started window takes no new chunk past this
	stopWithin      = 10 * time.Minute // ... nor this close to its reset
	weekUnderPace   = 10.0             // the week must be this many points under even pace: leftover 5h quota still counts against it
	weekLen         = 7 * 24 * time.Hour
)

// Inputs is what the loop reads before each decision.
type Inputs struct {
	Enabled     bool
	Quota       *claudequota.Quota
	Now         time.Time
	OwnRunsLive int // top-level runs not DUM-E's, neither done nor cancelled
	DumeRunLive bool
	Pressure    jobqueue.Pressure
	ArmedReset  int64 // the 5h reset (epoch s) of the window DUM-E has started; 0 = none
	HasWork     bool  // a chunk waits for DUM-E
}

// Decision is the gate's answer. ArmedReset is what the loop keeps for the next tick.
type Decision struct {
	Start      bool
	ArmedReset int64
	Reason     string
}

func Decide(in Inputs) Decision {
	if !in.Enabled {
		return Decision{Reason: "off"}
	}
	q := in.Quota
	if q == nil || q.FiveHourPct == nil || q.FiveHourReset == nil {
		return Decision{Reason: "no 5h quota reading"}
	}
	reset := *q.FiveHourReset
	left := time.Unix(reset, 0).Sub(in.Now)
	if left <= 0 {
		return Decision{Reason: "the 5h window has reset; waiting for a new reading"}
	}
	used := *q.FiveHourPct
	armed := in.ArmedReset == reset
	keep := int64(0)
	if armed {
		keep = reset
		if used >= stopAtUsedPct {
			return Decision{ArmedReset: keep, Reason: fmt.Sprintf("5h window %.0f%% used: done for this window", used)}
		}
		if left <= stopWithin {
			return Decision{ArmedReset: keep, Reason: fmt.Sprintf("%s to reset: done for this window", fmtDur(left))}
		}
	} else {
		if left > startWithin {
			return Decision{Reason: fmt.Sprintf("%s to the 5h reset; starts within %s", fmtDur(left), fmtDur(startWithin))}
		}
		if used > startMaxUsedPct {
			return Decision{Reason: fmt.Sprintf("5h window %.0f%% used; starts with at least %.0f%% left", used, 100-startMaxUsedPct)}
		}
	}
	wpct, wreset, ok := weekReading(q)
	if !ok {
		return Decision{ArmedReset: keep, Reason: "no weekly reading"}
	}
	if pace := weekPace(in.Now, wreset); wpct > pace-weekUnderPace {
		return Decision{ArmedReset: keep, Reason: fmt.Sprintf("week %.0f%% used at %.0f%% pace; needs %.0f points under", wpct, pace, weekUnderPace)}
	}
	if in.OwnRunsLive > 0 {
		return Decision{ArmedReset: keep, Reason: fmt.Sprintf("%d of your runs live", in.OwnRunsLive)}
	}
	if in.Pressure == jobqueue.PressureWarn || in.Pressure == jobqueue.PressureCritical {
		return Decision{ArmedReset: keep, Reason: "memory pressure high"}
	}
	// past here the window is DUM-E's: it stays armed whether or not a chunk starts now
	if in.DumeRunLive {
		return Decision{ArmedReset: reset, Reason: "working"}
	}
	if !in.HasWork {
		return Decision{ArmedReset: reset, Reason: "no chunk owned by dum-e waiting"}
	}
	return Decision{Start: true, ArmedReset: reset, Reason: "starting"}
}

// weekReading is the account's week, else its highest model-scoped week (a Team seat has no all-model one).
func weekReading(q *claudequota.Quota) (pct float64, reset int64, ok bool) {
	if q.WeekPct != nil && q.WeekReset != nil {
		return *q.WeekPct, *q.WeekReset, true
	}
	for _, w := range q.ScopedWeeks {
		if w.Reset != nil && (!ok || w.Pct > pct) {
			pct, reset, ok = w.Pct, *w.Reset, true
		}
	}
	return pct, reset, ok
}

// weekPace is the share of the week gone at now, in percent: what an even spend would have used.
func weekPace(now time.Time, reset int64) float64 {
	start := time.Unix(reset, 0).Add(-weekLen)
	p := 100 * float64(now.Sub(start)) / float64(weekLen)
	return min(100, max(0, p))
}

// fmtDur is "40 min" under an hour, else "2h05".
func fmtDur(d time.Duration) string {
	m := int(d.Round(time.Minute) / time.Minute)
	if m < 60 {
		return fmt.Sprintf("%d min", m)
	}
	return fmt.Sprintf("%dh%02d", m/60, m%60)
}
```

```go
// Copyright 2026, Command Line Inc.
// SPDX-License-Identifier: Apache-2.0

package dume

import (
	"fmt"
	"strconv"
	"strings"
	"time"

	"github.com/wavetermdev/waveterm/pkg/waveobj"
)

// Owner is the chunk owner that hands a chunk to DUM-E (`wsh effort chunk owner <effort> <chunk> dum-e`).
const Owner = "dum-e"

// DUM-E's notes on a chunk are its whole state: the newest one says where the chunk is.
const (
	notePrefix  = "DUM-E: "
	startedHead = notePrefix + "started run "
	toLandHead  = notePrefix + "ready to land "
	nothingHead = notePrefix + "nothing committed on "
)

func StartedNote(runId, branch string) string { return startedHead + runId + " on " + branch }
func ToLandNote(branch string, commits int) string {
	return fmt.Sprintf("%s%s (%d commits): git merge %s", toLandHead, branch, commits, branch)
}
func NothingNote(branch string) string {
	return nothingHead + branch + "; set the chunk back to pending to try again"
}

// ParseStarted reads a StartedNote.
func ParseStarted(text string) (runId, branch string, ok bool) {
	rest, found := strings.CutPrefix(text, startedHead)
	if !found {
		return "", "", false
	}
	runId, branch, found = strings.Cut(rest, " on ")
	return runId, branch, found && runId != "" && branch != ""
}

type State int

const (
	StateNone     State = iota // not DUM-E's, or a person took it (active with no DUM-E note)
	StateWaiting               // pending: in line
	StateRunning               // its run is live
	StateFinished              // its run ended; the loop has not checked the branch yet
	StateToLand                // committed on its branch, waiting for the person to land it
	StateBlocked
	StateDone
)

// ClassifyChunk is where a chunk stands for DUM-E. runLive says whether a run id is still going (neither done nor
// cancelled); a run that no longer exists is not live.
func ClassifyChunk(c waveobj.EffortChunk, runLive func(runId string) bool) State {
	if c.Owner != Owner {
		return StateNone
	}
	switch c.Status {
	case "pending":
		return StateWaiting
	case "done":
		return StateDone
	case "blocked":
		return StateBlocked
	case "active":
	default:
		return StateNone
	}
	note := newestNote(c)
	if strings.HasPrefix(note, toLandHead) {
		return StateToLand
	}
	if runId, _, ok := ParseStarted(note); ok {
		if runLive(runId) {
			return StateRunning
		}
		return StateFinished
	}
	return StateNone
}

func newestNote(c waveobj.EffortChunk) string {
	for i := len(c.Notes) - 1; i >= 0; i-- {
		if strings.HasPrefix(c.Notes[i].Text, notePrefix) {
			return c.Notes[i].Text
		}
	}
	return ""
}

// Ref names one DUM-E chunk.
type Ref struct {
	EffortOID   string
	EffortTitle string
	Project     string
	Chunk       string // the chunk's label
	RunId       string // from its newest started note
	Branch      string // ditto, or the to-land note's
	Notes       []string
}

type Summary struct {
	Waiting  []Ref // in order: the first is next
	Running  []Ref
	Finished []Ref
	ToLand   []Ref
	Blocked  int
	Done     int
}

// Summarize reads DUM-E's queue off the initiatives: only active ones, and a waiting chunk only when its
// initiative names a project (the run needs a checkout).
func Summarize(efforts []*waveobj.Effort, runLive func(runId string) bool) Summary {
	var s Summary
	for _, e := range efforts {
		if e == nil || e.Status != "active" {
			continue
		}
		for _, c := range e.Chunks {
			st := ClassifyChunk(c, runLive)
			if st == StateNone {
				continue
			}
			r := Ref{EffortOID: e.OID, EffortTitle: e.Title, Project: e.Project, Chunk: c.Label}
			for _, n := range c.Notes {
				r.Notes = append(r.Notes, n.Text)
			}
			if id, br, ok := ParseStarted(lastStarted(c)); ok {
				r.RunId, r.Branch = id, br
			}
			switch st {
			case StateWaiting:
				if e.Project != "" {
					s.Waiting = append(s.Waiting, r)
				}
			case StateRunning:
				s.Running = append(s.Running, r)
			case StateFinished:
				s.Finished = append(s.Finished, r)
			case StateToLand:
				s.ToLand = append(s.ToLand, r)
			case StateBlocked:
				s.Blocked++
			case StateDone:
				s.Done++
			}
		}
	}
	return s
}

func lastStarted(c waveobj.EffortChunk) string {
	for i := len(c.Notes) - 1; i >= 0; i-- {
		if strings.HasPrefix(c.Notes[i].Text, startedHead) {
			return c.Notes[i].Text
		}
	}
	return ""
}

// Goal is the quick run's goal for a chunk.
func Goal(r Ref) string {
	var b strings.Builder
	fmt.Fprintf(&b, "DUM-E chore for the initiative %q, chunk %q.\n", r.EffortTitle, r.Chunk)
	if len(r.Notes) > 0 {
		b.WriteString("\nThe chunk's notes:\n")
		for _, n := range r.Notes {
			b.WriteString("- " + n + "\n")
		}
	}
	b.WriteString("\nDo the chunk in this checkout and commit on its branch. Do not merge, push or land, and do not " +
		"change the chunk's status: the person lands it. Do not spawn subagents or workflows. If the chunk needs a " +
		"decision only the person can make, write the question as a chunk note (`wsh effort chunk note`) and finish " +
		"without committing.\n")
	return b.String()
}

// TreeKey is the worktree key (and, as wave/<key>, the branch) of one attempt at a chunk.
```

```ts
// The Brief header's DUM-E line: what DUM-E last decided, or the chunk it works, and how many of its branches wait
// for you to land them. Shown only while dume:enabled is on (the setting, read live, is the switch; the status can
// lag it by a tick). Pure.

export function dumeLine(s: DumeStatusData | null, enabled: boolean): string | null {
    if (!enabled || !s) {
        return null;
    }
    const head = s.running ? `working: ${s.running.chunk}` : s.reason || "not checked yet";
    const toLand = s.toland?.length ?? 0;
    return `DUM-E · ${head}${toLand > 0 ? ` · ${toLand} to land` : ""}`;
}
```
