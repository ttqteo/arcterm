# Fork a session from Chat Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Chat can fork its Claude session two ways: Fork (the whole conversation) in the agent header's ⋯ menu, and
Fork from here on any of your messages or the agent's replies (a hover button beside Copy, and the right-click menu).

**Architecture:** wavesrv gets one wshrpc command, `ForkAgentSessionCommand`, that turns a transcript path (plus,
for Fork from here, a record uuid and which side of it to keep) into a `claude --resume "<path>" --fork-session`
command line and the project path to launch it in. For Fork from here it first writes the kept prefix of the `.jsonl`
to `~/.arc/forks/<random>/<session id>.jsonl` and resumes that copy. The frontend tags each projected user message and
agent reply with its record uuid, carries it onto Chat's items, and launches the command through the existing
`launchAgent`, the same path History's Fork takes.

**Tech Stack:** Go (`pkg/agentsessions`, `pkg/wshrpc`), React 19 + jotai (`frontend/app/view/agents`), vitest, CDP
scenarios (`scripts/cdp/scenarios.mjs`).

## Decisions this plan builds on

- **What a fork from a message keeps.** From one of your messages: every record before it, and its text goes into the
  fork's chat box unsent, so you can edit it and send (Claude Code's rewind, as a new session). Pasted images are not
  carried. From an agent reply: every record up to and including it. The session you forked from is never touched.
- **How a Chat item points at its transcript record.** By the record's `uuid`, never a line number: Chat reads only
  the transcript's last window (`transcriptread.ts` reads 2000 lines, `livetranscript.ts` a tail), so a line index in
  the frontend is not a line of the file. Every Claude `user`, `assistant` and queued-prompt `attachment` record
  carries a `uuid`.
- **How the transcript is cut.** Go writes the lines before (or through) that record, unchanged, to
  `~/.arc/forks/<random>/<source session id>.jsonl` and runs `claude --resume "<copy>" --fork-session` in the source's
  project folder. A line prefix keeps the `parentUuid` chain valid. Copies older than a day are removed on the next fork.
  **Tried on 2026-11-12, claude 2.1.322:** a 28-line prefix copy of a real session, placed in a scratch folder outside
  `~/.claude/projects`, resumed with `--fork-session -p`, quoted the session's first message word for word; the new
  session was written under `~/.claude/projects/<slug of the cwd>/`, and the copy was left unchanged.
- **The whole-session Fork resumes the transcript by path too** (`claude --resume "<transcript path>"`), not by id:
  an id is looked up in claude's default config dir, a path works for any account's dir.
- **Paths go quoted with forward slashes** (`"C:/Users/…/x.jsonl"`): the line runs in whatever shell the tab starts,
  and an unquoted backslash path breaks in bash. claude accepts forward slashes on Windows (the try above used them).
- **The header's Fork is an item in the ⋯ menu**, not a new button: the header already holds the Chat/Terminal toggle,
  Ship and Done. Only a Claude agent with a `.jsonl` transcript offers either Fork.

// Copyright 2026, Command Line Inc.

package agentsessions

import (
	"crypto/rand"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"strings"
	"time"
)

// forkCopyTTL is how long a cut copy stays: claude reads it once, when the fork starts.
const forkCopyTTL = 24 * time.Hour

// ForkResult is the command line that starts a fork and the folder to start it in.
type ForkResult struct {
	Command     string
	ProjectPath string
}

// forkRecord is the part of a transcript line a fork reads.
type forkRecord struct {
	Type string `json:"type"`
	UUID string `json:"uuid"`
	Cwd  string `json:"cwd"`
}

// CutTranscript returns the lines a fork from the record `uuid` keeps: those before it (keep "before", a message of
// yours) or up to and including it (keep "through", a reply). A line prefix keeps the parentUuid chain valid. It fails
// when the record is not in the lines or nothing said is left.
func CutTranscript(lines []string, uuid, keep string) ([]string, error) {
	if keep != "before" && keep != "through" {
		return nil, fmt.Errorf("keep must be before or through, got %q", keep)
	}
	at := -1
	for i, l := range lines {
		if !strings.Contains(l, uuid) {
			continue
		}
		var rec forkRecord
		if json.Unmarshal([]byte(l), &rec) == nil && rec.UUID == uuid {
			at = i
			break
		}
	}
	if at < 0 {
		return nil, fmt.Errorf("that message is no longer in the transcript")
	}
	end := at
	if keep == "through" {
		end = at + 1
	}
	kept := lines[:end]
	for _, l := range kept {
		var rec forkRecord
		if json.Unmarshal([]byte(l), &rec) == nil && (rec.Type == "user" || rec.Type == "assistant") {
			return kept, nil
		}
	}
	return nil, fmt.Errorf("nothing before this message to fork from")
}

// transcriptCwd is the folder the session ran in: the last record that names one.
func transcriptCwd(lines []string) string {
	for i := len(lines) - 1; i >= 0; i-- {
		var rec forkRecord
		if json.Unmarshal([]byte(lines[i]), &rec) == nil && rec.Cwd != "" {
			return rec.Cwd
		}
	}
	return ""
}

// quotedPath is a path as one shell word in any shell a tab starts: forward slashes, double quotes.
func quotedPath(p string) string {
	return `"` + filepath.ToSlash(p) + `"`
}

// pruneForkCopies removes the cut copies older than forkCopyTTL.
func pruneForkCopies(dir string, now time.Time) {
	entries, err := os.ReadDir(dir)
	if err != nil {
		return
	}
	for _, e := range entries {
		info, err := e.Info()
		if err == nil && now.Sub(info.ModTime()) > forkCopyTTL {
			os.RemoveAll(filepath.Join(dir, e.Name()))
		}
	}
}

// ForkCommand builds the command line that forks the Claude session at `path`: the whole of it when uuid is "", else
// the part CutTranscript keeps, written to ~/.arc/forks/<random>/<same file name> and resumed from there. The source
// transcript is never written.
func ForkCommand(path, uuid, keep, home string) (ForkResult, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return ForkResult{}, fmt.Errorf("reading the transcript: %w", err)
	}
	lines := strings.Split(strings.TrimRight(string(data), "\r\n"), "\n")
	from := path
	if uuid != "" {
		kept, err := CutTranscript(lines, uuid, keep)
		if err != nil {
			return ForkResult{}, err
		}
		lines = kept
		root := filepath.Join(home, ".arc", "forks")
		pruneForkCopies(root, time.Now())
		var b [8]byte
		if _, err := rand.Read(b[:]); err != nil {
			return ForkResult{}, err
		}
		dir := filepath.Join(root, hex.EncodeToString(b[:]))
		if err := os.MkdirAll(dir, 0o755); err != nil {
			return ForkResult{}, fmt.Errorf("making the fork's copy: %w", err)
		}
		from = filepath.Join(dir, filepath.Base(path))
		if err := os.WriteFile(from, []byte(strings.Join(kept, "\n")+"\n"), 0o644); err != nil {
			return ForkResult{}, fmt.Errorf("writing the fork's copy: %w", err)
		}
	}
	project := transcriptCwd(lines)
	return ForkResult{
		Command:     claudeResumeCommand(quotedPath(from), project, home) + " --fork-session",
		ProjectPath: project,
	}, nil
}
```

// Copyright 2026, Command Line Inc.
// SPDX-License-Identifier: Apache-2.0
//
// Chat's Fork and Fork from here, as pure decisions: which agents can fork, where a fork from an item cuts, and the
// launch that starts it. ForkAgentSessionCommand (wavesrv) cuts the transcript and builds the command line.

import type { LaunchAgentOpts } from "@/app/cockpit/cockpit-actions";
import type { AgentVM } from "./agentsviewmodel";
import type { AgentPart, ChatItem } from "./chatmodel";

/** Where a fork from one Chat item cuts: before your message (its text goes back in the fork's box) or through a reply. */
export type ForkPoint = { uuid: string; keep: "before"; draft: string } | { uuid: string; keep: "through" };

/** Whether Chat offers Fork for this agent: Claude only (the one harness with --fork-session), with a transcript. */
export function canForkInChat(agent: Pick<AgentVM, "agent" | "transcriptPath">): boolean {
    return agent.agent === "claude" && !!agent.transcriptPath?.endsWith(".jsonl");
}

/** A fork from one of your messages: everything before it, its text back in the box. None for a command or no uuid. */
export function forkPointOfYou(item: Extract<ChatItem, { kind: "you" }>): ForkPoint | null {
    if (!("text" in item) || !item.uuid) {
        return null;
    }
    return { uuid: item.uuid, keep: "before", draft: item.text };
}

/** A fork from a reply: everything up to and including it. */
export function forkPointOfProse(part: Extract<AgentPart, { kind: "prose" }>): ForkPoint | null {
    return part.uuid ? { uuid: part.uuid, keep: "through" } : null;
}

function folderName(p: string): string {
    return p.split(/[\\/]/).filter(Boolean).pop() ?? "";
}

/** The launch that starts a fork: a new tab of its own (no resumeOf), named as a fork of the agent's task. */
export function chatForkLaunchOpts(
    agent: Pick<AgentVM, "task" | "project">,
    rtn: { command: string; projectpath: string }
): LaunchAgentOpts {
    const title = agent.task?.trim();
    return {
        runtime: "claude",
        startupCommand: rtn.command,
        task: "",
        projectPath: rtn.projectpath,
        projectName: agent.project || folderName(rtn.projectpath) || "agent",
        label: title ? `${title} (fork)` : "Fork",
    };
}
```

```ts
// Copyright 2026, Command Line Inc.
// SPDX-License-Identifier: Apache-2.0

import { launchAgent } from "@/app/cockpit/cockpit-actions";
import { pushToast } from "@/app/cockpit/notificationstore";
import { RpcApi } from "@/app/store/wshclientapi";
import { TabRpcClient } from "@/app/store/wshrpcutil";
import { fireAndForget } from "@/util/util";
import type { AgentsViewModel, AgentVM } from "./agentsviewmodel";
import { chatForkLaunchOpts, type ForkPoint } from "./chatfork";
import { setWakeDraft } from "./wakingstore";

/** Forks the agent's session (the whole of it, or from `point`) into a new tab; a fork from your message opens with
 *  that message's text in its box. A failure shows as a toast. */
export function forkFromChat(model: AgentsViewModel, agent: AgentVM, point?: ForkPoint): void {
    fireAndForget(async () => {
        try {
            const rtn = await RpcApi.ForkAgentSessionCommand(TabRpcClient, {
                transcriptpath: agent.transcriptPath!,
                ...(point ? { uuid: point.uuid, keep: point.keep } : {}),
            });
            const id = await launchAgent(model, chatForkLaunchOpts(agent, rtn));
            if (point?.keep === "before" && point.draft) {
                setWakeDraft(id, point.draft);
            }
        } catch (e) {
            pushToast({
                title: `Couldn't fork ${agent.name}`,
                message: e instanceof Error ? e.message : String(e),
                level: "error",
            });
        }
    });
}
```
